REM CK42X DeepSeek Shell Agent launcher - gated execution mode
REM Downloads the tagged GitHub PS1 to TEMP, verifies SHA256, then runs prompt chooser.
REM Execution still requires per-command confirmation and allowlist approval.
REM Requires local user env var: DEEPSEEK_API_KEY

DELAY 1000
GUI r
DELAY 300
STRING powershell.exe -NoProfile -ExecutionPolicy Bypass -Command "$u='https://raw.githubusercontent.com/lordbuffcloud/ck42x-ops-key/v0.2.0/scripts/ck42x-deepseek-shell.ps1';$s='9607ce4cb26f2af09cfbffbca952bc9adcd1f8f3b68c27f7587f959c944a6f73';$p=Join-Path $env:TEMP 'ck42x-deepseek-shell.ps1';Invoke-WebRequest -UseBasicParsing $u -OutFile $p;$h=(Get-FileHash $p -Algorithm SHA256).Hash.ToLower();if($h -ne $s){throw 'CK42X script SHA256 mismatch'};& $p -ChoosePrompt -Execute"
ENTER
